function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheets()[0];
  const values = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    sheetName: ss.getName() + ' / ' + sheet.getName(),
    rows: values
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const body = JSON.parse(e.postData.contents || '{}');
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheets()[0];
    const headers = ['id','date','colA','colB','colC','colD','taxType','taxFreeAmount','taxableAmount','amount','colE'];
    if (sheet.getLastRow() === 0) sheet.appendRow(headers);

    if (body.action === 'replaceAll' || body.action === 'syncAll' || body.action === 'clearAll') {
      sheet.clearContents();
      sheet.appendRow(headers);
      const list = Array.isArray(body.orders) ? body.orders : [];
      if (list.length > 0) {
        const rows = list.map(o => headers.map(k => o[k] !== undefined && o[k] !== null ? o[k] : ''));
        sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: list.length })).setMimeType(ContentService.MimeType.JSON);
    }

    if (body.action === 'appendOrders' && Array.isArray(body.orders) && body.orders.length > 0) {
      const rows = body.orders.map(o => headers.map(k => o[k] !== undefined && o[k] !== null ? o[k] : ''));
      sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, headers.length).setValues(rows);
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', added: rows.length })).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'noop' })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
